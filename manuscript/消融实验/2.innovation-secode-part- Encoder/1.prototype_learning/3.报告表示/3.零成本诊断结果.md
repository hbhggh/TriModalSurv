## 3. 解读（不改变判定）

1. **L1 / L2 的方向与假设相反。** 「证据不足」是规则给出的档位。但 TV(残缺) 的 CI 整段在正侧，所以在这批模型上，漏写了核心字段的报告仍然有用；合并后，它们帮的忙也不比完整报告少。
2. **逐癌描述（未预注册）**：只有 UCEC 符合假设方向（残缺 −0.021，完整 +0.058），每组只有 18 个事件。四个癌种里只有 1 个同向，与随机波动相容，不作为任何后续选项的依据。
3. **中心敏感性（按预注册只报主统计量方向）**：只用同中心的病人对时，TV(残缺) = +0.019，仍为正，方向与「残缺报告有害」相反。
4. **L5：没有达到「重要」门槛的尺度偏移。** 缺 RNA 时 G 的 CI 下限刚过 0（+0.0010）；两种模态 G 的 CI 上限（0.012、0.016）都低于 0.02，可以排除达到门槛的偏移。缺文本时，Mean 补偿下的 G（+0.015）高于不补偿（+0.006），但两者之差没有算 CI。
5. **缺失类型**：L1 / L2 的「残缺」是报告天然漏写，属于自然缺失；L5 的缺失状态全部来自人工遮挡。

## 4. 规则复盘与去向

STRATS_L1 = ["m1", "m0real"]                      # 主口径 m1，次口径 m0real
STRATS_L5 = ["m0real", "m1", "combo-l1-a1-w0p5"]  # No comp. / Mean / Proto.

SPEC_RE = re.compile(r"Specimen:\s*(.*?)(?=\.\s+[A-Z][A-Za-z /\-]{2,40}:|$)", re.S)


def gt(a, b):
    """a_i 相对 b_j：大于记 1，相等记 0.5。"""
    d = a[:, None] - b[None, :]
    return (d > 0) + 0.5 * (d == 0)


def cell(arm, c, s, sc, st):
    z = np.load(f"{ARMS[arm]}/raw/formal_eval/runs/test/{c}/{s}/{sc}/{st}.npz")
    return (z["patient_ids"].astype(str), z["risk_b"].astype(np.float64),
            z["time"].astype(np.float64), (z["censorship"] == 0).astype(np.float64))


def is_biopsy(text):
    m = SPEC_RE.search(text)
    if not m:
        return False
    s = m.group(1).lower()
    if "ectomy" in s:
        return False
    if "transurethral" in s or re.search(r"\btur", s):
        return True
    return ("biops" in s or "curettage" in s) and "resection" not in s


def build(c, info, core):
    ids, _, t, e = cell("B2", c, "123", "none", "m1")
    order = {p: i for i, p in enumerate(ids)}
    n = len(ids)
    risks = {}
    for arm in ARMS:
        for s in SEEDS:
            for sc in SCENARIOS:
                for st in sorted(set(STRATS_L1 + STRATS_L5)):
                    pid, r, tt, ee = cell(arm, c, s, sc, st)
                    assert sorted(pid) == sorted(ids), ("病人集合不一致", arm, c, s, sc, st)
                    ix = np.array([order[p] for p in pid])
                    rr, t2, e2 = np.empty(n), np.empty(n), np.empty(n)
                    rr[ix], t2[ix], e2[ix] = r, tt, ee
                    assert np.allclose(t2, t) and np.array_equal(e2, e), ("生存标签不一致", arm, c, s, sc, st)

    audit = json.load(open(f"{ARMS['B2']}/raw/formal_eval/runs/test/{c}/123/none/m1.audit.json"))

    nat_text = np.array([bool(nv[p]["text"]) for p in ids])
    nat_rna = np.array([bool(nv[p]["rna"]) for p in ids])

             comp=comp, comp_same=comp * same)
    if c in C_L1:
        ok = np.zeros(n, bool)
        inc = np.zeros(n, bool)
        biopsy = np.zeros(n, bool)
        for i, p in enumerate(ids):
            r = info.get(p)
            if r is None or r.get("status") != "ok":
                continue
            ok[i] = True
            biopsy[i] = is_biopsy(r["text"])
            req = [f for f in core[c] if not (biopsy[i] and f in NA_BIOPSY)]
            inc[i] = any((f + ":") not in r["text"] for f in req)
        pop1 = ok & nat_text
        pop2 = pop1 & nat_rna
        d.update(ok=ok, inc=inc, biopsy=biopsy, pop1=pop1, pop2=pop2)
        # 各 seed 一致性先取均值，再乘可比对
        K = {}
        for arm in ARMS:
            for st in STRATS_L1:
                for sc in SCENARIOS:
                    K[(arm, st, sc)] = np.mean([gt(risks[(arm, st, sc, s)], risks[(arm, st, sc, s)])

        D = {}

def q(v, M):
    return float(v @ M @ v)


def l12_stats(data, W, arm, st, stat, pop_key, same=False, drop=None, cancers=None):
    """返回 {组: 合并值}，组 = inc / com；分子分母跨纳入癌种相加。"""
    out = {}
    for g in ["inc", "com"]:
        num = den = 0.0
        for c in cancers:
            if c == drop:
                continue
            d = data[c]
            mask = d[pop_key] & (d["inc"] if g == "inc" else ~d["inc"])
            v = W[c] * mask
            Dm = (d["D_same"] if same else d["D"])[(arm, st, stat)]
            Cm = d["comp_same"] if same else d["comp"]
            num += q(v, Dm)
            den += q(v, Cm)
        out[g] = num / den if den > 0 else np.nan
    out["diff"] = out["com"] - out["inc"]
    return out


def zscore_cross(r, v, half):
    """交叉拟合 z 标准化：每一半用另一半（按权重 v）的均值与标准差。"""
    z = np.empty_like(r)
    for h in (True, False):
        other = (half != h)
        w = v * other
        sw = w.sum()
        mu = (w * r).sum() / sw
        sd = np.sqrt((w * (r - mu) ** 2).sum() / sw)
        z[half == h] = (r[half == h] - mu) / sd
    return z


def l5_percancer(d, v, arm, mod, st):
    num_raw = q(v, d["Xraw"][(arm, mod, st)])
    xs = []
    for s in SEEDS:
        zf = zscore_cross(d["risks"][(arm, st, "none", s)], v, d["half"])
        zm = zscore_cross(d["risks"][(arm, st, f"{mod}_100", s)], v, d["half"])
        xs.append(0.5 * (gt(zf, zm) + gt(zm, zf)))
    num_al = q(v, np.mean(xs, axis=0) * d["comp"])
    den = q(v, d["comp"])
    return num_raw, num_al, den


def l5_stats(data, W, arm, mod, st):
    per = {}
    tot_raw = tot_al = tot_den = 0.0
    for c in C_L5:
        d = data[c]
        pop = d["nat_text"] if mod == "text" else d["nat_rna"]
        v = W[c] * pop
        nr, na, dn = l5_percancer(d, v, arm, mod, st)
        per[c] = (na - nr) / dn
        tot_raw += nr
        tot_al += na
        tot_den += dn
    return dict(G=(tot_al - tot_raw) / tot_den, C_raw=tot_raw / tot_den, C_al=tot_al / tot_den, per=per)


def ci(xs):
    lo, hi = np.nanpercentile(np.asarray(xs, float), [2.5, 97.5])
    return float(lo), float(hi)

        print(f"  {c}: 核心 {len(core[c])} 个 {core[c]}")
        print(f"       test {row['n_test']}，L1 人群 {row['n_pop1']}（活检类 {row['n_biopsy']}）；"
              f"残缺 {row['n_inc']} 人 / {row['ev_inc']} 事件，完整 {row['n_com']} 人 / {row['ev_com']} 事件")
    inc_L1 = tuple(c for c in C_L1 if desc[c]["ev_inc"] >= MIN_EV_CANCER and desc[c]["ev_com"] >= MIN_EV_CANCER)
    inc_L2 = tuple(c for c in C_L1 if desc[c]["ev_inc_L2"] >= MIN_EV_CANCER and desc[c]["ev_com_L2"] >= MIN_EV_CANCER)
    ev_pool = {"L1": (sum(desc[c]["ev_inc"] for c in inc_L1), sum(desc[c]["ev_com"] for c in inc_L1)),
               "L2": (sum(desc[c]["ev_inc_L2"] for c in inc_L2), sum(desc[c]["ev_com_L2"] for c in inc_L2))}
    print(f"  纳入合并：L1 {inc_L1}（事件 残缺 {ev_pool['L1'][0]} / 完整 {ev_pool['L1'][1]}）；"
          f"L2 {inc_L2}（事件 残缺 {ev_pool['L2'][0]} / 完整 {ev_pool['L2'][1]}）")

    # 2. 点估计
    ones = {c: np.ones(data[c]["n"]) for c in C_L5}
    L12_KEYS = []
    for arm in ARMS:
        for st in STRATS_L1:
            L12_KEYS += [(arm, st, "TV", "pop1", False, inc_L1), (arm, st, "TVR", "pop2", False, inc_L2),
                         (arm, st, "TV", "pop2", False, inc_L2),  # 额外损害用 L2 人群的 TV
                         (arm, st, "TV", "pop1", True, inc_L1), (arm, st, "TVR", "pop2", True, inc_L2)]
    point = {k: l12_stats(data, ones, k[0], k[1], k[2], k[3], same=k[4], cancers=k[5]) for k in L12_KEYS}
    L5_KEYS = [(arm, mod, st) for arm in ARMS for mod in ["text", "rna"] for st in STRATS_L5]
    point5 = {k: l5_stats(data, ones, *k) for k in L5_KEYS}
    loco = {}
    for (arm, st, stat, pk, cancers) in [("B2", "m1", "TV", "pop1", inc_L1), ("B2", "m1", "TVR", "pop2", inc_L2)]:
        loco[(stat)] = {c: l12_stats(data, ones, arm, st, stat, pk, drop=c, cancers=cancers) for c in cancers}
    percancer = {}
    for c in C_L1:
        percancer[c] = {st: {stat: l12_stats(data, ones, "B2", st, stat, pk, cancers=[c])
                             for stat, pk in [("TV", "pop1"), ("TVR", "pop2")]} for st in STRATS_L1}

    # 3. bootstrap（按癌种分层，同一组权重贯穿所有口径）
    rng = np.random.default_rng(SEED)
    boot = collections.defaultdict(list)
    boot5 = collections.defaultdict(list)
    for b in range(B):
        W = {c: np.bincount(rng.integers(0, data[c]["n"], data[c]["n"]), minlength=data[c]["n"]).astype(float)
             for c in C_L5}
        for k in L12_KEYS:
            r = l12_stats(data, W, k[0], k[1], k[2], k[3], same=k[4], cancers=k[5])
            for g in ["inc", "com", "diff"]:
                boot[(k, g)].append(r[g])
        for k in L5_KEYS:
            if k[0] != "B2":
                continue
            boot5[k].append(l5_stats(data, W, *k)["G"])
        if (b + 1) % 100 == 0:
            print(f"  bootstrap {b + 1}/{B}", flush=True)

    # 4. 输出与判定
    res = dict(B=B, core=core, desc=desc, included=dict(L1=inc_L1, L2=inc_L2), events_pooled=ev_pool,
               L12={}, L5={}, loco={}, percancer={}, verdict={})
    print("\n[L1/L2 结果] 值 = 合并点估计，括号 = 95% CI")
    for k in L12_KEYS:
        arm, st, stat, pk, same, _ = k
        tag = f"{arm} {st:6s} {stat:3s} {pk}{' 同中心' if same else ''}"
        r = point[k]
        cis = {g: ci(boot[(k, g)]) for g in ["inc", "com", "diff"]}
        res["L12"][tag] = {g: dict(est=r[g], ci=cis[g]) for g in ["inc", "com", "diff"]}
        print(f"  {tag:28s} 残缺 {r['inc']:+.4f} ({cis['inc'][0]:+.4f}, {cis['inc'][1]:+.4f})  "
              f"完整 {r['com']:+.4f} ({cis['com'][0]:+.4f}, {cis['com'][1]:+.4f})  "
              f"完整−残缺 {r['diff']:+.4f} ({cis['diff'][0]:+.4f}, {cis['diff'][1]:+.4f})")
    print("\n[L1/L2 各癌种点估计，B2]")
    for c in C_L1:
        for st in STRATS_L1:
            pc = percancer[c][st]
            print(f"  {c} {st:6s} TV 残缺 {pc['TV']['inc']:+.4f} 完整 {pc['TV']['com']:+.4f} | "
                  f"TV_rna 残缺 {pc['TVR']['inc']:+.4f} 完整 {pc['TVR']['com']:+.4f}")
        res["percancer"][c] = {st: {stat: {g: percancer[c][st][stat][g] for g in ["inc", "com", "diff"]}
                                    for stat in ["TV", "TVR"]} for st in STRATS_L1}
    print("\n[留一癌种，B2 m1]")
    for stat in loco:
        for c, r in loco[stat].items():
            print(f"  {stat} 去掉 {c}: 残缺 {r['inc']:+.4f}  完整−残缺 {r['diff']:+.4f}")
        res["loco"][stat] = {c: {g: r[g] for g in ["inc", "diff"]} for c, r in loco[stat].items()}

    def verdict(stat, pk, cancers, evkey):
        k1 = ("B2", "m1", stat, pk, False, cancers)
        k0 = ("B2", "m0real", stat, pk, False, cancers)
        est, est0 = point[k1], point[k0]
        ci_inc, ci_diff = ci(boot[(k1, "inc")]), ci(boot[(k1, "diff")])
        ev_i, ev_c = ev_pool[evkey]
        hw = (ci_inc[1] - ci_inc[0]) / 2
        sign_ok = np.sign(est["inc"]) == np.sign(est0["inc"])
        loco_ok = all(np.sign(r["inc"]) == np.sign(est["inc"]) and np.sign(r["diff"]) == np.sign(est["diff"])
                      for r in loco[stat].values())
        reasons = dict(ev_inc=ev_i, ev_com=ev_c, half_width=hw, m1_m0real_same_sign=bool(sign_ok),
                       loco_signs_unchanged=bool(loco_ok), ci_inc=ci_inc, ci_diff=ci_diff)
        if min(ev_i, ev_c) < MIN_EV_POOLED or hw > MAX_HALF_WIDTH or not sign_ok or np.isnan(est["inc"]):
            return "证据不足", reasons
        if est["inc"] < 0 and ci_inc[1] < 0 and ci_diff[0] > 0 and loco_ok:
            return "成立（发现）", reasons
        if ci_diff[0] > 0:
            return "预期信息损失（不推进）", reasons
        return "判否", reasons

    for name, stat, pk, cancers, evkey in [("L1", "TV", "pop1", inc_L1, "L1"), ("L2", "TVR", "pop2", inc_L2, "L2")]:
        v, why = verdict(stat, pk, cancers, evkey)
        res["verdict"][name] = dict(verdict=v, reasons=why)
        print(f"\n[判定] {name}：{v}  依据 {why}")
    e1 = point[("B2", "m1", "TVR", "pop2", False, inc_L2)]["inc"] - point[("B2", "m1", "TV", "pop2", False, inc_L2)]["inc"]
    print(f"  L2 描述：额外损害 TV_rna(残缺) − TV(残缺) = {e1:+.4f}（B2 m1，L2 人群）")
    res["L2_extra_B2_m1"] = e1

    print("\n[L5 结果] G = 对齐后 − 原始 的跨状态一致性（合并）；B2 带 95% CI，B1 只报点估计")
    for k in L5_KEYS:
        arm, mod, st = k
        p = point5[k]
        cis = ci(boot5[k]) if arm == "B2" else (np.nan, np.nan)
        res["L5"][f"{arm} {mod} {st}"] = dict(G=p["G"], ci=cis, C_raw=p["C_raw"], C_al=p["C_al"], per=p["per"])
        per = " ".join(f"{c}{v:+.3f}" for c, v in p["per"].items())
        print(f"  {arm} 缺{mod:4s} {st:16s} C_cross 原始 {p['C_raw']:.4f} → 对齐 {p['C_al']:.4f}  "
              f"G {p['G']:+.4f} ({cis[0]:+.4f}, {cis[1]:+.4f})  各癌 {per}")
    for mod in ["text", "rna"]:
        p = point5[("B2", mod, "m0real")]
        cis = ci(boot5[("B2", mod, "m0real")])
        hw = (cis[1] - cis[0]) / 2
        npos = sum(v > 0 for v in p["per"].values())
        cis_m1 = ci(boot5[("B2", mod, "m1")])
        third = "第 3 条不过（现有 Mean 补偿已解决）" if cis_m1[0] <= 0 <= cis_m1[1] else "第 3 条待实验（Mean 补偿下 G 的 CI 不含 0）"
        if hw > MAX_HALF_WIDTH:
            v = "证据不足"
        elif p["G"] >= 0.02 and cis[0] > 0 and npos >= 4:
            v = "成立（发现）；" + third
        else:
            v = "判否"
        res["verdict"][f"L5-{mod}"] = dict(verdict=v, G=p["G"], ci=cis, half_width=hw, n_pos=int(npos), ci_m1=cis_m1)
        print(f"[判定] L5 缺{mod}：{v}  G {p['G']:+.4f} CI ({cis[0]:+.4f}, {cis[1]:+.4f}) 半宽 {hw:.4f}，"
              f"为正癌种 {npos}/5；Mean 下 CI ({cis_m1[0]:+.4f}, {cis_m1[1]:+.4f})")
    if OUT:
        json.dump(res, open(OUT, "w"), ensure_ascii=False, indent=1, default=float)
        print(f"\n结果已写入 {OUT}")


if __name__ == "__main__":
    main()
